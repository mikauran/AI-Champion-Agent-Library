export interface Channel { send(msg: unknown): void; onMessage(cb: (m: unknown) => void): void; onNoise(cb: (line: string) => void): void; onClose(cb: (info: {code?: number|null; signal?: string|null}) => void): void; close(): Promise<void>; stderrText?(): string }
export type FindingKind = 'stdout_noise'|'oversize_message'|'unknown_protocol_version'|'timeout'|'bad_response'|'invalid_json_rpc'|'early_exit'|'session_error'
export interface Finding { kind: FindingKind; detail: string }   // detail <= 200 chars
export type InitResult = { ok: true; protocolVersion: string; serverInfo: {name?: string; version?: string}; capabilities: unknown } | { ok: false; kind: 'timeout'|'closed'|'error'|'bad_response'; reason: string }
export const KNOWN_PROTOCOL_VERSIONS = ['2025-11-25','2025-06-18','2025-03-26','2024-11-05']
export class McpClient implements ToolCaller {

  tools: ToolInfo[]; findings: Finding[]; close(): Promise<void>; stderrText(): string
  call(...)  // alias of callTool to satisfy ToolCaller
}
