```ts
// behaviour.ts: knobs; GOOD is the default object, variants override a subset
export interface Behaviour {
  mirror: boolean                 // 1:1 endpoint tools only, raw payload, no clamp, no derived tool, leaky/bare errors
  derivedTool: boolean            // expose find_items_by_attribute (client-side filter over pages)
  projection: boolean             // cap top-level fields at 12 (id, code, name, category, supplier, attributes + selected scalars)
  clamp: boolean                  // clamp per_page to documented max (50) and say so
  errorStyle: 'clean'|'leaky'|'bare-empty'|'no-auth-message'
  retry: boolean                  // retry 429/5xx up to 3 attempts, honour Retry-After, injectable sleep (default backoff 20 ms)
  malformed: 'honest'|'fabricate'
  writeTool: boolean              // adds delete_item (POST/DELETE upstream attempt)
  echoSecret: boolean             // includes ACME_API_KEY value in a tool output and on stderr
  grounded: boolean               // false = also calls an undocumented route
  validateCodes: boolean          // false = forwards invalid GTIN to upstream
  largeResponse: 'bounded'|'huge'|'silent-cap'
  processFault?: 'crash-on-init'|'hang'|'stdout-noise'|'old-protocol'
}
export interface VariantDef { id: string; description: string; behaviour: Behaviour; mustFail: string[]; flags: FlagId[]; expectScoreBand?: 'full'|'partial'|'low' }
export function serveStdio(input: Readable, output: Writable, opts: { variant: string; env: Record<string,string>; sleep?: (ms:number)=>Promise<void> }): Promise<void>
export function startHttpFake(opts: { variant: string; env: Record<string,string> }): Promise<{ url: string; close(): Promise<void> }>
```
