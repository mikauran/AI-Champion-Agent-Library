The slug is derived, NOT declared (`src/lib/spec/types.ts` lines 43-49):
```typescript
export function slugify(input: string): string {
  return input.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/-{2,}/g, '-').replace(/^-|-$/g, '')
}
```
